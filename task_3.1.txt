<script>
let display = document.getElementById("display");
let buttons = document.querySelectorAll("button");

let input = "";
let isDegree = true; // Default degree mode

buttons.forEach(button => {
    button.addEventListener("click", () => {
        let value = button.textContent;

        if (value === "C") {
            input = "";
            display.value = "";
        }

        else if (value === "=") {
            try {
                let expression = input
                    .replace(/π/g, "Math.PI")
                    .replace(/e/g, "Math.E")
                    .replace(/√/g, "Math.sqrt")
                    .replace(/\^/g, "**")
                    .replace(/log/g, "Math.log10")
                    .replace(/ln/g, "Math.log");

                // Degree conversion for sin, cos, tan
                expression = expression.replace(/sin\((.*?)\)/g,
                    (_, num) => `Math.sin(${num} * Math.PI / 180)`);

                expression = expression.replace(/cos\((.*?)\)/g,
                    (_, num) => `Math.cos(${num} * Math.PI / 180)`);

                expression = expression.replace(/tan\((.*?)\)/g,
                    (_, num) => `Math.tan(${num} * Math.PI / 180)`);

                input = eval(expression);
                display.value = input;
            } catch {
                display.value = "Error";
            }
        }

        else {
            input += value;
            display.value = input;
        }
    });
});
</script>
